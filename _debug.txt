# /uapi <api_name> [参数...]
        api_name = parts[0]
        api = API_MAP.get(api_name)

        if not api:
            suggestions = self._fuzzy_find(api_name)
            if suggestions:
                yield event.plain_result(
                    f"❓ 未找到 API '{api_name}'。\n\n你可能想找：\n" +
                    "\n".join(f"  • {s}" for s in suggestions[:10]) +
                    f"\n\n使用 /uapi search {api_name} 搜索更多。"
                )
            else:
                yield event.plain_result(
                    f"❓ 未找到 API '{api_name}'。\n使用 /uapi list 查看所有可用 API。"
                )
            return

        # 如果没有提供参数，显示 API 详情
        if len(parts) == 1:
            yield event.plain_result(self._format_api_info(api))
            return

        # 解析参数: 自动判断是 key=value 模式还是位置参数模式
        arg_values = parts[1:]
        has_keyword = any("=" in part for part in arg_values)

        if has_keyword:
            # 传统 key=value 模式
            params = self._parse_keyword_args(arg_values)
            result = await self._call_api_and_format_legacy(event, api, params)
            yield result
        else:
            # 位置参数模式（简化模式）
            result = await self._call_api_positional(event, api, arg_values)
            yield result
